cmake_minimum_required(VERSION 3.20)
project(modloader_sandbox C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(WIN32)
  set(vendorDefault ON)
else()
  set(vendorDefault OFF)
endif()
option(wheatVendorLua "Build Lua 5.4.8 from source instead of using the system one" ${vendorDefault})
set(wheatProxy "winmm" CACHE STRING "DLL the loader impersonates (needs loader/proxy/<name>.def and <name>Exports.inc)")

if(wheatVendorLua)
  include(cmake/LuaVendored.cmake)
else()
  find_package(PkgConfig QUIET)
  set(luaFound FALSE)
  if(PkgConfig_FOUND)
    foreach(modName lua5.4 lua54 lua-5.4)
      pkg_check_modules(LUA54 QUIET IMPORTED_TARGET ${modName})
      if(LUA54_FOUND)
        set(luaTarget PkgConfig::LUA54)
        set(luaFound TRUE)
        message(STATUS "Using Lua 5.4 via pkg-config (${modName})")
        break()
      endif()
    endforeach()
  endif()
  if(NOT luaFound)
    find_package(Lua 5.4 EXACT REQUIRED)
    add_library(lua54Fallback INTERFACE)
    target_include_directories(lua54Fallback INTERFACE ${LUA_INCLUDE_DIR})
    target_link_libraries(lua54Fallback INTERFACE ${LUA_LIBRARIES})
    set(luaTarget lua54Fallback)
  endif()
endif()

if(MINGW)
  add_link_options(-static)
endif()

set(strlibDir third_party/lua_strlib)
set_source_files_properties(${strlibDir}/lstrlib_patched.c PROPERTIES
  COMPILE_DEFINITIONS "_CRT_SECURE_NO_WARNINGS;_XOPEN_SOURCE=600;_LARGEFILE_SOURCE;_FILE_OFFSET_BITS=64"
  LANGUAGE C)
add_library(lua_sandbox STATIC src/lua_sandbox.cpp ${strlibDir}/lstrlib_patched.c)
target_include_directories(lua_sandbox PUBLIC include PRIVATE ${strlibDir})
target_link_libraries(lua_sandbox PUBLIC ${luaTarget})
if(MSVC)
  target_compile_options(lua_sandbox PRIVATE /W4)
else()
  target_compile_options(lua_sandbox PRIVATE -Wall -Wextra)
endif()

add_library(mod_host STATIC loader/mod_host.cpp loader/manifest.cpp loader/modData.cpp loader/consoleApi.cpp)
target_include_directories(mod_host PUBLIC include loader)
target_link_libraries(mod_host PUBLIC lua_sandbox)

if(UNIX AND NOT APPLE)
  set(imguiDir ${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  set(wheatImGuiSources
    ${imguiDir}/imgui.cpp
    ${imguiDir}/imgui_draw.cpp
    ${imguiDir}/imgui_tables.cpp
    ${imguiDir}/imgui_widgets.cpp)
  target_include_directories(mod_host PUBLIC ${imguiDir})
  target_sources(mod_host PRIVATE loader/imguiApi.cpp)
endif()

if(MINGW)
  configure_file(loader/proxy/${wheatProxy}Exports.inc ${CMAKE_BINARY_DIR}/gen/proxyExports.inc COPYONLY)
  add_library(loaderProxy SHARED loader/proxyMain.cpp loader/frameHook.cpp loader/engineConsole.cpp loader/proxy/${wheatProxy}.def)
  target_include_directories(loaderProxy PRIVATE ${CMAKE_BINARY_DIR}/gen)
  set_target_properties(loaderProxy PROPERTIES OUTPUT_NAME wheatWindows PREFIX "")
  target_link_libraries(loaderProxy PRIVATE mod_host)
  target_link_options(loaderProxy PRIVATE -static-libgcc -static-libstdc++ -s -Wl,--no-insert-timestamp)
elseif(UNIX AND NOT APPLE)
  # Linux Steam loads this ELF preload library under the platform-specific build name.
  find_package(Vulkan REQUIRED)
  add_library(loaderProxy SHARED loader/proxyMainLinux.cpp loader/engineConsoleLinux.cpp loader/frameHookLinux.cpp ${wheatImGuiSources} ${imguiDir}/backends/imgui_impl_sdl2.cpp)
  target_include_directories(loaderProxy PRIVATE ${imguiDir} ${imguiDir}/backends)
  set_target_properties(lua_sandbox mod_host PROPERTIES POSITION_INDEPENDENT_CODE ON)
  set_target_properties(loaderProxy PROPERTIES OUTPUT_NAME wheatLinux PREFIX "" SUFFIX ".dll")
  target_link_libraries(loaderProxy PRIVATE mod_host dl pthread PkgConfig::SDL2)
  add_library(wheatVulkanLayer SHARED loader/vulkanLayerLinux.cpp)
  target_sources(wheatVulkanLayer PRIVATE ${imguiDir}/backends/imgui_impl_vulkan.cpp)
  target_include_directories(wheatVulkanLayer PRIVATE ${Vulkan_INCLUDE_DIRS} ${imguiDir} ${imguiDir}/backends)
  set_target_properties(wheatVulkanLayer PROPERTIES OUTPUT_NAME wheatVulkanLayer PREFIX "lib" SUFFIX ".so")
  target_link_libraries(wheatVulkanLayer PRIVATE dl pthread Vulkan::Vulkan)
endif()
